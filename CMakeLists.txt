include(${CMAKE_SOURCE_DIR}/cmake/pr_common.cmake)

option(CONFIG_USE_LUAJIT "Use LuaJIT." ON)

set(PROJ_NAME luasystem)
pr_add_library(${PROJ_NAME} SHARED)
pr_add_export_macro(${PROJ_NAME} DLLLUA)

if (WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
	# WINDOWS_CLANG_COMPILER_FIX
	target_compile_definitions(${PROJ_NAME} PUBLIC
		"SCRIPT_DIRECTORY=GET_SCRIPT_DIRECTORY()"
		"SCRIPT_DIRECTORY_SLASH=GET_SCRIPT_DIRECTORY_SLASH()"
		"FILE_EXTENSION=GET_FILE_EXTENSION()"
		"FILE_EXTENSION_PRECOMPILED=GET_FILE_EXTENSION_PRECOMPILED()"
		"DOT_FILE_EXTENSION=GET_DOT_FILE_EXTENSION()"
		"DOT_FILE_EXTENSION_PRECOMPILED=GET_DOT_FILE_EXTENSION_PRECOMPILED()"

		"LIB_COROUTINE=GET_LIB_COROUTINE()"
		"LIB_MATH=GET_LIB_MATH()"
		"LIB_STRING=GET_LIB_STRING()"
		"LIB_TABLE=GET_LIB_TABLE()"
		"LIB_IO=GET_LIB_IO()"
		"LIB_OS=GET_LIB_OS()"
		"LIB_PACKAGE=GET_LIB_PACKAGE()"
		"LIB_DEBUG=GET_LIB_DEBUG()"
		"LIB_BIT=GET_LIB_BIT()"
		"LIB_JIT=GET_LIB_JIT()"
		"LIB_FFI=GET_LIB_FFI()"
	)
endif()

if(CONFIG_USE_LUAJIT)
	pr_add_compile_definitions(${PROJ_NAME} -DUSE_LUAJIT PUBLIC)
	set(LUAJIT_DISABLE_JIT OFF CACHE BOOL "Disable the JIT compiler" FORCE)
else()
	set(LUAJIT_DISABLE_JIT ON CACHE BOOL "Disable the JIT compiler" FORCE)
endif()

pr_add_dependency(${PROJ_NAME} vfilesystem TARGET PUBLIC)
pr_add_third_party_dependency(${PROJ_NAME} luajit PUBLIC)
pr_add_dependency(${PROJ_NAME} luabind TARGET PUBLIC)

pr_init_module(${PROJ_NAME})

pr_add_compile_definitions(${PROJ_NAME} -DLUABIND_DYNAMIC_LINK PUBLIC)
pr_add_compile_definitions(${PROJ_NAME} -DDLLLUA_EX)

pr_finalize(${PROJ_NAME})
